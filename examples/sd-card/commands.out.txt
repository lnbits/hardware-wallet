# SD-card output reference. Do not rely on edits to this file: the device
# recreates commands.out.txt at startup whenever commands.in.txt is detected.
# Only responses that can be written by an SD-card command are shown below.
# Angle-bracketed text is a placeholder; actual output contains real values.

#### commands output ####

# Wallet creation, restore, and replacement
/create 1
/create 0
/restore 1
/restore 0
/wipe 1
/wipe 0

# Authentication
/password 1
/password 0
/password-clear 1

# Public wallet data
/xpub 1 <extended-public-key> <master-fingerprint>
/xpub 0 invalid_mnemonic
/address 1 <derived-address>

# /seed displays the mnemonic only on the hardware screen and writes no output
# to this file.

# Visual hardware-RNG check. Only aggregate results are written; raw RNG values
# and individual histogram bins remain on the hardware display.
/trng 1 5000 <chi-squared> <minimum-bin-count> <maximum-bin-count> <healthy-or-unexpected>
/trng 0 <reason>

# PSBT acceptance or failure
/psbt 1
/psbt psbt_parse_failed
/psbt invalid_mnemonic
/psbt review_rejected

# A successful /psbt writes one block per transaction output and its fee. The
# device requires a physical approval for every output, the fee, and the final
# signing action. Output indexes start at 0.
Output <index>
Address: <address>
Amount: <amount-in-satoshis> sat
Fee: <fee-in-satoshis> sat
/sign <signed-input-count> <signed-base64-PSBT>
